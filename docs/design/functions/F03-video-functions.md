### 函数
```typescript
/**
 * 生成视频
 * @param request - 生成请求
 * @returns Promise<VideoResponse>
 */
async function generateVideo(request: VideoRequest): Promise<VideoResponse>

interface VideoRequest {
  prompt: string           // 提示词
  duration?: number        // 时长（秒），默认5
  resolution?: string      // 分辨率 "720p" | "1080p"
  provider?: string        // 提供商
}

interface VideoResponse {
  success: boolean
  taskId?: string          // 任务ID（用于查询进度）
  videoUrl?: string        // 视频URL（完成后）
  localPath?: string       // 本地路径
  status: 'pending' | 'processing' | 'completed' | 'failed'
  progress?: number        // 进度 0-100
}
```

### 函数
```typescript
/**
 * 图片生成视频
 * @param request - 图生视频请求
 * @returns Promise<VideoResponse>
 */
async function imageToVideo(request: ImageToVideoRequest): Promise<VideoResponse>

interface ImageToVideoRequest {
  imagePath: string         // 图片路径
  prompt?: string           // 运动描述（可选）
  duration?: number         // 时长
  resolution?: string       // 分辨率
}
```

## F03-04 生成进度

### 功能概述
| 属性 | 值 |
|-----|-----|
| 功能编号 | F03-04 |
| 功能名称 | 生成进度 |
| 优先级 | P0 |
| 描述 | 显示视频生成进度 |

### 函数
```typescript
/**
 * 查询生成状态
 * @param taskId - 任务ID
 * @returns Promise<VideoStatus>
 */
async function getVideoStatus(taskId: string): Promise<VideoStatus>

interface VideoStatus {
  status: 'pending' | 'processing' | 'completed' | 'failed'
  progress: number          // 0-100
  estimatedTime?: number    // 预计剩余时间（秒）
  error?: string            // 错误信息
}
```
