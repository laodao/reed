# 剧集制作 · Phase 3c · 360 全景 + 视角截图 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 给 episode-production 模块上线 spec §4.5.10 描述的"360 全景 + 视角截图"机制——AI 生成 / 上传 panorama_360 全景图、浏览器内 360 拖动预览（photo-sphere-viewer）、用户调好 yaw/pitch/fov 截 2D 图作为分镜背景 ref，让同场景多分镜背景在 3D 空间中天然一致。

**Architecture:** 截图机制采用前端渲染（photo-sphere-viewer canvas → toDataURL → POST 给后端）+ 后端落盘的混合方案，避开后端"等距投影 → 透视图"数学渲染。新增 2 个 service（panorama-generator / panorama-snapshot），1 个前端组件 PanoramaViewer 封装 photo-sphere-viewer，1 个 PanoramaSnapshotEditor 含视角选择器。LocationGalleryEditor（P3a task 13 已建）加新区接入。storyboard 加 background_ref_image_id 字段（spec §4.8）让 P5 渲染域消费。

**Tech Stack:** TypeScript / Vue 3 + Element Plus / [photo-sphere-viewer](https://photo-sphere-viewer.js.org/) v5 (~80KB) / sharp (后端写入图片，已在依赖) / TypeORM / Vitest + Playwright。

**前置依赖**：P3a 完成（assets/services/* + asset-image.service / location-asset.service / LocationGalleryEditor 已就绪）。

```typescript
/**
 * storyboards 加 background_ref_image_id 字段（spec §4.8）
 * 指向 asset_image.id（category='panorama_snapshot'），关联 360 截图作分镜背景 ref

 * 兼容 SQLite + PG，幂等
 */

import type { DataSource } from 'typeorm'

export async function addStoryboardBackgroundRef(ds: DataSource): Promise<void> {
  const isPg = ds.options.type === 'postgres'

  if (!await checkTableExists(ds, 'storyboards', isPg)) return

  if (await checkColumnExists(ds, 'storyboards', 'background_ref_image_id', isPg)) return

  const colType = isPg ? 'varchar' : 'TEXT'
  await ds.query(`ALTER TABLE storyboards ADD COLUMN background_ref_image_id ${colType}`)
  console.log('[add-storyboard-background-ref] storyboards: 加列 background_ref_image_id')
}

async function checkTableExists(ds: DataSource, table: string, isPg: boolean): Promise<boolean> {
  if (isPg) {
    const r = await ds.query(`SELECT 1 FROM information_schema.tables WHERE table_name=$1 AND table_schema=current_schema() LIMIT 1`, [table])
    return r.length > 0
  }
  const r = await ds.query(`SELECT 1 FROM sqlite_master WHERE type='table' AND name=? LIMIT 1`, [table])
  return r.length > 0
}

async function checkColumnExists(ds: DataSource, table: string, column: string, isPg: boolean): Promise<boolean> {
  if (isPg) {
    const r = await ds.query(`SELECT 1 FROM information_schema.columns WHERE table_name=$1 AND column_name=$2 AND table_schema=current_schema() LIMIT 1`, [table, column])
    return r.length > 0
  }
  const r = await ds.query(`SELECT 1 FROM pragma_table_info(?) WHERE name=? LIMIT 1`, [table, column])
  return r.length > 0
}
```

```typescript
/**
 * panorama 生成 service（spec §4.5.10 step 1）
 * AI 生成 360 全景或用户上传，写入 asset_image (category='panorama_360')
 * @module modules/episode-production/assets/services/panorama-generator
 */
import type { AssetImageService } from './asset-image.service'

export interface PanoramaImageGenerator {
  generate(opts: {
    prompt: string
    negativePrompt?: string
    aspectRatio: '2:1'
  }): Promise<{ imagePath: string; thumbnailPath: string | null }>
}

export interface GenerateForInput {
  locationBindingId: string
  stylePrompt: string             // 地点风格描述（"茶馆 古风 工笔国风"）
  additionalPrompt?: string
  negativePrompt?: string
  userId?: string | null
}

export interface UploadForInput {
  locationBindingId: string
  imagePath: string
  userId?: string | null
}

const PANORAMA_PROMPT_SUFFIX = ' equirectangular panoramic 360 view 2:1 aspect ratio'

export class PanoramaGeneratorService {
  constructor(
    private readonly imageSvc: AssetImageService,
    private readonly imageGen: PanoramaImageGenerator,
  ) {}

  async generateFor(input: GenerateForInput) {
    const fullPrompt = input.stylePrompt + (input.additionalPrompt ? ' ' + input.additionalPrompt : '') + PANORAMA_PROMPT_SUFFIX
    const result = await this.imageGen.generate({
      prompt: fullPrompt,
      negativePrompt: input.negativePrompt,
      aspectRatio: '2:1',
    })

    return await this.imageSvc.addImage({
      assetType: 'location',
      assetId: input.locationBindingId,
      category: 'panorama_360',
      imagePath: result.imagePath,
      thumbnailPath: result.thumbnailPath ?? undefined,
      isMain: false,
      prompt: fullPrompt,
      negativePrompt: input.negativePrompt,
      userId: input.userId,
    })
  }

  async uploadFor(input: UploadForInput) {
    return await this.imageSvc.addImage({
      assetType: 'location',
      assetId: input.locationBindingId,
      category: 'panorama_360',
      imagePath: input.imagePath,
      isMain: false,
      userId: input.userId,
    })
  }

  async listForLocation(locationBindingId: string) {
    return await this.imageSvc.listByCategory('location', locationBindingId, 'panorama_360')
  }

  async deletePanorama(panoramaImageId: string) {
    await this.imageSvc.deleteImage(panoramaImageId)
  }
}
```

```typescript
/**
 * panorama 视角截图 service（spec §4.5.10 step 3）
 * 接收前端 photo-sphere-viewer 截的 base64 图 + 视角参数，落盘 + 写 asset_image
 * @module modules/episode-production/assets/services/panorama-snapshot
 */
import { randomBytes } from 'crypto'
import type { AssetImageService } from './asset-image.service'

export interface Viewpoint {
  yaw: number    // -180 ~ 180
  pitch: number  // -90 ~ 90
  fov: number    // 30 ~ 120
}

export interface SnapshotFileWriter {
  /** 把 base64 图存到目录，返回完整路径 */
  writeBase64Image(opts: { base64: string; targetDir: string; fileName: string }): Promise<string>
}

export interface CreateSnapshotInput {
  locationBindingId: string
  sourcePanoramaImageId: string  // 来源 panorama_360 的 asset_image.id
  viewpoint: Viewpoint
  base64Image: string             // 前端 canvas toDataURL 的 base64（可含 data:image/jpeg;base64, 前缀）
  userId?: string | null
}

export class PanoramaSnapshotService {
  constructor(
    private readonly imageSvc: AssetImageService,
    private readonly writer: SnapshotFileWriter,
    private readonly snapshotsDir: string,
  ) {}

  async createSnapshot(input: CreateSnapshotInput) {
    const id = `snap_${randomBytes(8).toString('hex')}`
    const fileName = `snapshot-${id}.jpg`

    // 去掉 base64 data URL 前缀
    const cleanBase64 = input.base64Image.replace(/^data:image\/[a-z]+;base64,/, '')

    const fullPath = await this.writer.writeBase64Image({
      base64: cleanBase64,
      targetDir: this.snapshotsDir,
      fileName,
    })

    return await this.imageSvc.addImage({
      assetType: 'location',
      assetId: input.locationBindingId,
      category: 'panorama_snapshot',
      imagePath: fullPath,
      isMain: false,
      label: `yaw${input.viewpoint.yaw}_pitch${input.viewpoint.pitch}_fov${input.viewpoint.fov}`,
      metadata: {
        yaw: input.viewpoint.yaw,
```

**Concept:** 封装 photo-sphere-viewer 的最小可用组件——传入 panorama imagePath，渲染 360 拖动预览，暴露 ref 让父组件能调 `getCurrentViewpoint()` / `captureCanvas()`。

```vue
<script setup lang="ts">
import { ref, onMounted, onBeforeUnmount, watch } from 'vue'
import { Viewer } from '@photo-sphere-viewer/core'
import '@photo-sphere-viewer/core/index.css'

const props = defineProps<{
  imagePath: string
  height?: number
  initialYaw?: number
  initialPitch?: number
  initialFov?: number
}>()

const emit = defineEmits<{
  'viewpoint-change': [viewpoint: { yaw: number; pitch: number; fov: number }]
}>()

const container = ref<HTMLElement | null>(null)
let viewer: Viewer | null = null

onMounted(() => {
  if (!container.value) return

  viewer = new Viewer({
    container: container.value,
    panorama: props.imagePath,
    defaultYaw: degToRad(props.initialYaw ?? 0),
    defaultPitch: degToRad(props.initialPitch ?? 0),
    defaultZoomLvl: fovToZoomLvl(props.initialFov ?? 70),
    navbar: ['zoom', 'fullscreen'],
  })

  viewer.addEventListener('position-updated', () => {
    emit('viewpoint-change', getCurrentViewpoint())
  })
  viewer.addEventListener('zoom-updated', () => {
    emit('viewpoint-change', getCurrentViewpoint())
  })
})

onBeforeUnmount(() => {
  viewer?.destroy()
  viewer = null
})

watch(() => props.imagePath, (newPath) => {
  viewer?.setPanorama(newPath)
})

function degToRad(deg: number): number { return deg * Math.PI / 180 }
function radToDeg(rad: number): number { return rad * 180 / Math.PI }
function fovToZoomLvl(fovDeg: number): number {
  // photo-sphere-viewer 用 0-100 zoom level，对应 fov ~120° (zoom=0) 到 ~30° (zoom=100)
  return Math.max(0, Math.min(100, (120 - fovDeg) / 0.9))
}
function zoomLvlToFov(zoom: number): number {
  return 120 - zoom * 0.9
}

function getCurrentViewpoint(): { yaw: number; pitch: number; fov: number } {
  if (!viewer) return { yaw: 0, pitch: 0, fov: 70 }
  const pos = viewer.getPosition()
  const zoom = viewer.getZoomLevel()
  return {
    yaw: Math.round(radToDeg(pos.yaw) * 100) / 100,
    pitch: Math.round(radToDeg(pos.pitch) * 100) / 100,
    fov: Math.round(zoomLvlToFov(zoom) * 100) / 100,
  }
}

async function captureCanvas(): Promise<string> {
  if (!viewer) throw new Error('Viewer not initialized')
  const canvas = container.value!.querySelector('canvas') as HTMLCanvasElement
  if (!canvas) throw new Error('Canvas not found')
  return canvas.toDataURL('image/jpeg', 0.92)
}

defineExpose({ getCurrentViewpoint, captureCanvas })
</script>

</style>
```

```typescript
import type { DataSource } from 'typeorm'
import { StoryboardEntity } from '@core/database/entities/episode-production/storyboard.entity'

export class StoryboardService {
  constructor(private readonly ds: DataSource) {}
  private get repo() { return this.ds.getRepository(StoryboardEntity) }

  async setBackgroundRefImage(storyboardId: string, imageId: string | null): Promise<void> {
    await this.repo.update(storyboardId, { backgroundRefImageId: imageId, updatedAt: new Date() } as any)
  }

  async getBackgroundRefImageId(storyboardId: string): Promise<string | null> {
    const sb = await this.repo.findOne({ where: { id: storyboardId } })
    return sb?.backgroundRefImageId ?? null
  }
}
```
